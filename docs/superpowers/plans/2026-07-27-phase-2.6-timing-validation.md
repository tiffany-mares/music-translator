```python
"""Phase 2.6: timing breakdown for one chunked-pipeline execution.

Usage: python scripts/timing_report.py <execName> [--song <songId>]
Requires AWS credentials in the environment (aws configure export-credentials).

Prints per-state durations from the Step Functions execution history, the
per-chunk SageMaker job breakdown (provision = Started-Created, run =
Ended-Started, from DescribeProcessingJob), and the comparison against
architecture.md's ~70-110 s pipeline target.
"""
import json
import sys

REGION = "us-east-1"
ACCOUNT = "503233513399"
BUCKET = "lyralearn-audio-503233513399"


def state_durations(events: list) -> dict:
    entered, durations = {}, {}
    for ev in events:
        if ev["type"].endswith("StateEntered"):
            entered[ev["stateEnteredEventDetails"]["name"]] = ev["timestamp"]
        elif ev["type"].endswith("StateExited"):
            name = ev["stateExitedEventDetails"]["name"]
            if name in entered:
                durations[name] = (ev["timestamp"] - entered[name]).total_seconds()
    return durations


def summarize_jobs(jobs: list) -> dict:
    provisions = [(j["started"] - j["created"]).total_seconds() for j in jobs]
    runs = [(j["ended"] - j["started"]).total_seconds() for j in jobs]
    return {
        "count": len(jobs),
        "avg_provision_s": sum(provisions) / len(provisions),
        "max_provision_s": max(provisions),
        "avg_run_s": sum(runs) / len(runs),
        "max_run_s": max(runs),
    }


def main() -> int:
    import boto3

    exec_name = sys.argv[1]
    song_id = sys.argv[sys.argv.index("--song") + 1] if "--song" in sys.argv else "test-song-001"

    sfn = boto3.client("stepfunctions", region_name=REGION)
    s3 = boto3.client("s3", region_name=REGION)
    sm = boto3.client("sagemaker", region_name=REGION)

    arn = f"arn:aws:states:{REGION}:{ACCOUNT}:execution:lyralearn-pipeline-chunked:{exec_name}"
    desc = sfn.describe_execution(executionArn=arn)
    total = (desc["stopDate"] - desc["startDate"]).total_seconds()

    events, token = [], None
    while True:
        kwargs = {"executionArn": arn, "maxResults": 500}
        if token:
            kwargs["nextToken"] = token
        page = sfn.get_execution_history(**kwargs)
        events.extend(page["events"])
        token = page.get("nextToken")
        if not token:
            break

    durations = state_durations(events)

    manifest = json.loads(s3.get_object(
        Bucket=BUCKET, Key=f"songs/{song_id}/chunks/{exec_name}/manifest.json")["Body"].read())
    jobs = []
    for item in manifest["chunks"]:
        d = sm.describe_processing_job(ProcessingJobName=item["jobName"])
        jobs.append({"jobName": item["jobName"], "created": d["CreationTime"],
                     "started": d["ProcessingStartTime"], "ended": d["ProcessingEndTime"]})
    js = summarize_jobs(jobs)

    print(f"execution {exec_name} ({song_id}, {manifest['songDurationSeconds']:.1f}s song, "
          f"{manifest['chunkCount']} chunks): status={desc['status']}")
    print(f"  TOTAL WALL: {total:.1f}s   (target: 70-110s)")
    for name in ("MarkProcessing", "ChunkAudio", "ProcessChunksInParallel",
                 "StitchResults", "RunTranslation", "MarkComplete"):
        if name in durations:
            print(f"  {name:<24} {durations[name]:8.1f}s")
    print(f"  chunk jobs: {js['count']} | provision avg/max {js['avg_provision_s']:.0f}/"
          f"{js['max_provision_s']:.0f}s | run avg/max {js['avg_run_s']:.0f}/{js['max_run_s']:.0f}s")
    print(f"  divergence: total is {total / 110:.1f}x the target ceiling"
          if total > 110 else "  within target range")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
